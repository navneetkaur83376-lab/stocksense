Add these two blocks to your existing index.js, right next to where productRoutes is wired up:

const warehouseRoutes = require('./routes/warehouses');
app.use('/api/warehouses', warehouseRoutes(db));

const receiptRoutes = require('./routes/receipts');
app.use('/api/receipts', receiptRoutes(db));

const transferRoutes = require('./routes/transfers');
app.use('/api/transfers', transferRoutes(db));

That's it — no other changes needed to index.js.
